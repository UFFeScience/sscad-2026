Escreva um script em utilizando o sistema de Workflow Nextflow para gerar workflows responsáveis por calcular os algoritmos KMeans, Matmul e Lysozyme separadamente. A aplicação Lysozyme precisa do gmx e gmx_mpi do GROMACS com a topologia pdb2gmx. A seguir um exemplo de um workflow de sequenciamento de RNA que deve servir como referência.

### script1.nf
```groovy
/*
 * pipeline input parameters
 */
params.reads = "$projectDir/data/ggal/gut_{1,2}.fq"
params.transcriptome_file = "$projectDir/data/ggal/transcriptome.fa"
params.multiqc = "$projectDir/multiqc"
params.outdir = "results"

workflow {
    log.info """\
        R N A S E Q - N F   P I P E L I N E
        ===================================
        transcriptome: ${params.transcriptome_file}
        reads        : ${params.reads}
        outdir       : ${params.outdir}
        """.stripIndent(true)
}
```

### script4.nf
```groovy
/*
 * pipeline input parameters
 */
params.reads = "$projectDir/data/ggal/gut_{1,2}.fq"
params.transcriptome_file = "$projectDir/data/ggal/transcriptome.fa"
params.multiqc = "$projectDir/multiqc"
params.outdir = "results"

/*
 * define the `INDEX` process that creates a binary index
 * given the transcriptome file
 */
process INDEX {
    cpus 2

    input:
    path transcriptome

    output:
    path 'salmon_index'

    script:
    """
    salmon index --threads $task.cpus -t $transcriptome -i salmon_index
    """
}

process QUANTIFICATION {
    tag "Salmon on $sample_id"
    publishDir params.outdir, mode: 'copy'

    input:
    path index
    tuple val(sample_id), path(reads)

    output:
    path "$sample_id"

    script:
    """
    salmon quant --threads $task.cpus --libType=U -i $index -1 ${reads[0]} -2 ${reads[1]} -o $sample_id
    """
}

workflow {
    log.info """\
        R N A S E Q - N F   P I P E L I N E
        ===================================
        transcriptome: ${params.transcriptome_file}
        reads        : ${params.reads}
        outdir       : ${params.outdir}
        """.stripIndent(true)

    channel
        .fromFilePairs(params.reads, checkIfExists: true)
        .set { read_pairs_ch }

    index_ch = INDEX(params.transcriptome_file)
    quant_ch = QUANTIFICATION(index_ch, read_pairs_ch)
}
```

### nextflow.config
```groovy
profiles {
    docker {
        docker.enabled = true
        process.container = 'nextflow/rnaseq-nf'
    }

    conda {
        conda.enabled = true
        process.conda = "$projectDir/conda.yml"
    }
}
```

### conda.yml
```yaml
name: rnaseq-nf
channels:
  - conda-forge
  - bioconda
dependencies:
  - bioconda::fastqc=0.12.1
  - bioconda::multiqc=1.25
  - bioconda::salmon=1.10.3
```

### bin/fastqc.sh
```bash
#!/bin/bash
set -e
set -u

sample_id=${1}
reads=${2}

mkdir fastqc_${sample_id}_logs
fastqc -o fastqc_${sample_id}_logs -f fastq -q ${reads}
```
